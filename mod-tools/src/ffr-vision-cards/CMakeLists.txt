cmake_minimum_required(VERSION 3.22)
project(FFRVisionCards LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 23)
set(CMAKE_MSVC_RUNTIME_LIBRARY MultiThreadedDLL)
get_filename_component(FFR_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../.." ABSOLUTE)
set(DEPS "${FFR_ROOT}/build/tooling")
set(SDK "${DEPS}/ue4ss-sdk-24b12662")
set(GENERATED "${FFR_ROOT}/build/abilities/native-bridge-sdk")
set(BUILD_SHARED_LIBS OFF CACHE BOOL "" FORCE)
add_subdirectory("${DEPS}/minhook-1.3.4" minhook EXCLUDE_FROM_ALL)
add_library(FFRVisionCards SHARED mod.cpp "${FFR_ROOT}/src/ffr-availability-native/bridge_core.cpp")
target_link_libraries(FFRVisionCards PRIVATE minhook "${GENERATED}/UE4SS.lib" bcrypt)
target_include_directories(FFRVisionCards PRIVATE "${FFR_ROOT}/src/ffr-availability-native")
target_include_directories(FFRVisionCards SYSTEM PRIVATE "${GENERATED}/include" "${SDK}/UE4SS/include"
  "${SDK}/deps/first/LuaMadeSimple/include" "${SDK}/deps/first/LuaRaw/include"
  "${SDK}/deps/first/File/include" "${SDK}/deps/first/Input/include" "${SDK}/deps/first/String/include"
  "${DEPS}/fmt-11.2.0/include")
target_compile_definitions(FFRVisionCards PRIVATE NOMINMAX WIN32_LEAN_AND_MEAN HAS_INPUT _DISABLE_CONSTEXPR_MUTEX_CONSTRUCTOR)
target_compile_options(FFRVisionCards PRIVATE /W4 /EHsc /utf-8)
set_target_properties(FFRVisionCards PROPERTIES OUTPUT_NAME main)
